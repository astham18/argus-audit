## template 0: Return the sum of all prices.
-- correct --
def total(prices):
    s = 0
    for p in prices:
        s += p
    return s
-- buggy (skips the first item) --
def total(prices):
    s = 0
    for p in prices[1:]:
        s += p
    return s

## template 1: Return True if the person is 18 or older.
-- correct --
def is_adult(age):
    return age >= 18
-- buggy (wrong comparison at 18) --
def is_adult(age):
    return age > 18

## template 2: Return the last item.
-- correct --
def last(items):
    return items[len(items) - 1]
-- buggy (index out of range) --
def last(items):
    return items[len(items)]

## template 3: Return the mean of the numbers.
-- correct --
def average(xs):
    return sum(xs) / len(xs)
-- buggy (wrong denominator) --
def average(xs):
    return sum(xs) / (len(xs) + 1)

## template 4: Count the even numbers.
-- correct --
def count_even(xs):
    return sum(1 for x in xs if x % 2 == 0)
-- buggy (counts odd numbers) --
def count_even(xs):
    return sum(1 for x in xs if x % 2 == 1)

## template 5: Return the largest number.
-- correct --
def maximum(xs):
    m = xs[0]
    for x in xs:
        if x > m:
            m = x
    return m
-- buggy (returns the minimum) --
def maximum(xs):
    m = xs[0]
    for x in xs:
        if x < m:
            m = x
    return m

## template 6: Return the area of a rectangle.
-- correct --
def area(w, h):
    return w * h
-- buggy (adds instead of multiplying) --
def area(w, h):
    return w + h

## template 7: Return the first n items.
-- correct --
def first_n(xs, n):
    return xs[:n]
-- buggy (off by one) --
def first_n(xs, n):
    return xs[:n - 1]

## template 8: Return True if v is in the list.
-- correct --
def contains(xs, v):
    for x in xs:
        if x == v:
            return True
    return False
-- buggy (returns after the first item) --
def contains(xs, v):
    for x in xs:
        if x == v:
            return True
        return False

## template 9: Convert Fahrenheit to Celsius.
-- correct --
def celsius(f):
    return (f - 32) * 5 / 9
-- buggy (wrong conversion factor) --
def celsius(f):
    return (f - 32) * 9 / 5
